Code Quality MCP Server

from mcp.server.fastmcp import FastMCP import ast mcp = FastMCP("code-quality") @mcp.tool() def analyze_complexity(file_path: str) -> str: """Analyze cyclomatic complexity of a Python file.""" with open(file_path) as f: tree = ast.parse(f.read()) results = [] for node in ast.walk(tree): if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)): complexity = count_branches(node) results.append(f"{node.name}: complexity={complexity}") return "\n".join(results) if results else "No functions found" @mcp.tool() def find_code_issues(file_path: str) -> str: """Find common code issues using AST analysis.""" with open(file_path) as f: tree = ast.parse(f.read()) issues = [] for node in ast.walk(tree): if isinstance(node, ast.ExceptHandler) and node.type is None: issues.append(f"Line {node.lineno}: Bare except clause") if isinstance(node, ast.FunctionDef) and len(node.args.args) > 5: issues.append(f"Line {node.lineno}: {node.name}() has {len(node.args.args)} params (>5)") return "\n".join(issues) if issues else "No issues found"

AST analysis is static — no code execution needed. Safe to run on untrusted code, fast, and catches structural issues that linters miss.

Test Runner MCP Server

import subprocess @mcp.tool() def run_tests(test_path: str = "tests/", verbose: bool = False) -> str: """Run pytest and return results summary.""" cmd = ["python", "-m", "pytest", test_path, "--tb=short"] if verbose: cmd.append("-v") result = subprocess.run(cmd, capture_output=True, text=True, timeout=60) return f"Exit code: {result.returncode}\n\n{result.stdout}\n{result.stderr}" @mcp.tool() def get_test_coverage(source_dir: str = "src/") -> str: """Run tests with coverage and return report.""" cmd = ["python", "-m", "pytest", f"--cov={source_dir}", "--cov-report=term-missing"] result = subprocess.run(cmd, capture_output=True, text=True, timeout=120) return result.stdout @mcp.tool() def find_failing_tests(test_path: str = "tests/") -> str: """Find and list only failing tests.""" cmd = ["python", "-m", "pytest", test_path, "--tb=line", "-q", "--no-header"] result = subprocess.run(cmd, capture_output=True, text=True, timeout=60) if result.returncode == 0: return "All tests passing!" return f"Failing tests:\n{result.stdout}"
Always set timeout on subprocess calls. A runaway test suite can hang the MCP server indefinitely.

Documentation Generator MCP Server

@mcp.tool() def generate_docs(file_path: str) -> str: """Extract docstrings and generate documentation for a Python file.""" with open(file_path) as f: tree = ast.parse(f.read()) docs = [] for node in ast.walk(tree): if isinstance(node, (ast.FunctionDef, ast.ClassDef)): docstring = ast.get_docstring(node) kind = "class" if isinstance(node, ast.ClassDef) else "function" docs.append(f"### {kind}: {node.name}\n{docstring or 'No docstring'}\n") return "\n".join(docs) if docs else "No classes or functions found" @mcp.tool() def check_missing_docs(file_path: str) -> str: """Find functions and classes missing docstrings.""" with open(file_path) as f: tree = ast.parse(f.read()) missing = [] for node in ast.walk(tree): if isinstance(node, (ast.FunctionDef, ast.ClassDef)): if not ast.get_docstring(node): missing.append(f"Line {node.lineno}: {node.name} — missing docstring") return "\n".join(missing) if missing else "All functions/classes are documented!" @mcp.tool() def create_readme(project_dir: str) -> str: """Generate a README.md skeleton from project structure.""" # Scan for setup.py/pyproject.toml, main modules, test dir import os files = os.listdir(project_dir) sections = ["# Project Name\n", "## Installation\n", "## Usage\n"] if "tests" in files: sections.append("## Testing\n```bash\npytest tests/\n```\n") return "\n".join(sections)

Code Review as a Graph

A code review is naturally a multi-step workflow with conditional branching

load_diff
→
analyze_code
→
check_tests
→
generate_review
↓ if issues found                      ↓ if approved
suggest_fixes
done
↓ feeds back to generate_review          
The conditional edge after generate_review is key: if severity is “request_changes”, the graph routes to suggest_fixes before completing.

Review Agent State

from typing import TypedDict, List class ReviewState(TypedDict): diff: str # Code changes to review file_analyses: List[dict] # Per-file analysis results test_results: str # Test execution output issues: List[dict] # Found issues review_summary: str # Final review text severity: str # overall: "approve" | "request_changes"
Field Set By Purpose
diff load_diff node Raw code changes from git or PR
file_analyses analyze_code node Complexity, issues per file (via MCP)
test_results check_tests node Pytest output and coverage (via MCP)
issues analyze_code node Categorized list of problems found
severity generate_review node Controls conditional routing in graph

Building the Review Workflow

from langgraph.graph import StateGraph, END workflow = StateGraph(ReviewState) # Add nodes workflow.add_node("analyze", analyze_code_node) workflow.add_node("test", run_tests_node) workflow.add_node("review", generate_review_node) workflow.add_node("suggest_fixes", suggest_fixes_node) # Add edges workflow.add_edge("analyze", "test") workflow.add_edge("test", "review") workflow.add_conditional_edges( "review", lambda state: "suggest" if state["severity"] == "request_changes" else "end", {"suggest": "suggest_fixes", "end": END} ) workflow.add_edge("suggest_fixes", END) # Set entry point and compile workflow.set_entry_point("analyze") graph = workflow.compile()

Node Implementations with MCP

from mcp import ClientSession async def analyze_code_node(state: ReviewState) -> dict: """Use MCP code-quality server to analyze changed files.""" analyses = [] issues = [] async with ClientSession(transport) as session: for file in extract_files(state["diff"]): # Call MCP tool: analyze_complexity complexity = await session.call_tool( "analyze_complexity", {"file_path": file} ) # Call MCP tool: find_code_issues file_issues = await session.call_tool( "find_code_issues", {"file_path": file} ) analyses.append({"file": file, "complexity": complexity}) if file_issues != "No issues found": issues.extend(parse_issues(file_issues)) return {"file_analyses": analyses, "issues": issues} async def run_tests_node(state: ReviewState) -> dict: """Use MCP test-runner server to execute tests.""" async with ClientSession(transport) as session: results = await session.call_tool("run_tests", {"verbose": True}) return {"test_results": results}

Multi-Agent Review

Specialized reviewers running in parallel, feeding into a synthesis step

Security Reviewer

  • Checks for injection vulnerabilities
  • Scans for hardcoded secrets & API keys
  • Validates authentication/authorization patterns
  • Flags unsafe deserialization

Performance Reviewer

  • Detects N+1 query patterns
  • Flags potential memory leaks
  • Checks algorithmic complexity (O(n²) loops)
  • Identifies missing caching opportunities

Style Reviewer

  • Validates naming conventions
  • Checks code organization & structure
  • Verifies docstring coverage
  • Enforces project-specific rules

Synthesis Node

  • Merges all reviewer outputs
  • Deduplicates overlapping findings
  • Ranks issues by severity
  • Produces final review summary

Multi-Agent Graph Structure

from langgraph.graph import StateGraph, END workflow = StateGraph(ReviewState) # Parallel reviewer nodes workflow.add_node("security_review", security_reviewer_node) workflow.add_node("performance_review", performance_reviewer_node) workflow.add_node("style_review", style_reviewer_node) # Synthesis node merges all reviews workflow.add_node("synthesize", synthesize_reviews_node) # Fan-out: entry runs all three in parallel workflow.set_entry_point("security_review") workflow.add_edge("security_review", "synthesize") workflow.add_edge("performance_review", "synthesize") workflow.add_edge("style_review", "synthesize") # Synthesize produces the final verdict workflow.add_edge("synthesize", END) graph = workflow.compile()
Each reviewer is an independent LangGraph node that can use MCP tools. The synthesis node aggregates findings into a unified review with ranked priorities.

Tool Registries & Sandboxed Execution

Discovery · Routing · Sandboxing · LangChain Bridge

Why Tool Registries?

As your tool collection grows, you need a way to manage, discover, and route

Without a Registry

  • Hardcoded tool lists in every agent
  • Adding a tool means changing agent code
  • No way to search for tools by capability
  • Duplicate tool implementations across projects

With a Registry

  • Dynamic tool discovery at runtime
  • Register new tools without touching agents
  • Search by tags, description, or category
  • Central place for versioning and access control
A tool registry is like a phone book for capabilities. Instead of memorizing every number (hardcoding tools), you look up what you need by name or category.

Tool Registry Pattern

from dataclasses import dataclass from typing import Callable @dataclass class ToolInfo: name: str handler: Callable description: str tags: list[str] class ToolRegistry: def __init__(self): self._tools: dict[str, ToolInfo] = {} def register(self, name: str, handler, description: str, tags: list[str]): self._tools[name] = ToolInfo(name, handler, description, tags) def discover(self, query: str = "", tags: list[str] = None) -> list[ToolInfo]: results = list(self._tools.values()) if tags: results = [t for t in results if set(tags) & set(t.tags)] if query: results = [t for t in results if query.lower() in t.description.lower()] return results def route(self, tool_name: str, **kwargs): if tool_name not in self._tools: raise ToolNotFoundError(f"Unknown tool: {tool_name}") return self._tools[tool_name].handler(**kwargs)

Using the Tool Registry

# Create registry and register tools registry = ToolRegistry() registry.register( name="analyze_complexity", handler=analyze_complexity, description="Analyze cyclomatic complexity of Python files", tags=["code-quality", "python", "analysis"] ) registry.register( name="run_tests", handler=run_tests, description="Execute pytest and return results", tags=["testing", "python", "pytest"] ) # Discover tools by tag python_tools = registry.discover(tags=["python"]) # Returns: [analyze_complexity, run_tests] # Discover tools by description search test_tools = registry.discover(query="test") # Returns: [run_tests] # Route a call dynamically result = registry.route("run_tests", test_path="tests/", verbose=True)

Sandboxed Execution

AI-generated commands must be constrained to prevent damage

Why Sandbox?

  • AI-generated code can be dangerous
  • Must prevent: file system damage
  • Must prevent: network abuse
  • Must prevent: infinite loops
  • Must prevent: resource exhaustion

How to Sandbox

  • Subprocess with timeout
  • Restricted PATH environment
  • Resource limits (memory, CPU)
  • Output size limits
  • Allowlisted commands only
Never execute AI-generated commands with full system access. Even seemingly harmless commands can cause data loss or security breaches when composed creatively.

Subprocess Sandbox Implementation

import subprocess def sandboxed_exec(command: str, timeout: int = 30, max_output: int = 10000) -> str: """Execute a command with safety constraints.""" BLOCKED = ["rm -rf", "sudo", "curl", "wget", "chmod 777"] for blocked in BLOCKED: if blocked in command: return f"BLOCKED: '{blocked}' is not allowed" try: result = subprocess.run( command, shell=True, capture_output=True, text=True, timeout=timeout, env={"PATH": "/usr/bin:/usr/local/bin"} # Restricted PATH ) output = result.stdout[:max_output] if result.returncode != 0: output += f"\nSTDERR: {result.stderr[:2000]}" return output except subprocess.TimeoutExpired: return f"TIMEOUT: Command exceeded {timeout}s limit"
Safety Layer What It Prevents
Blocklist Destructive commands (rm -rf, sudo, network tools)
Timeout Infinite loops and runaway processes
Restricted PATH Access to unauthorized binaries
Output limit Memory exhaustion from large outputs

Bridging MCP ↔ LangChain

Wrap MCP tools as LangChain StructuredTools for use in any agent

from langchain_core.tools import StructuredTool from mcp import ClientSession def mcp_to_langchain(session: ClientSession, tool_name: str, description: str) -> StructuredTool: """Bridge an MCP tool into a LangChain StructuredTool.""" async def call_mcp_tool(**kwargs) -> str: result = await session.call_tool(tool_name, kwargs) return str(result) return StructuredTool.from_function( func=call_mcp_tool, name=tool_name, description=description, coroutine=call_mcp_tool, ) # Usage: wrap MCP tools for a LangChain agent tools = [ mcp_to_langchain(session, "analyze_complexity", "Analyze cyclomatic complexity of a Python file"), mcp_to_langchain(session, "run_tests", "Run pytest and return results"), mcp_to_langchain(session, "find_code_issues", "Find common code issues using AST"), ] # Now use with any LangChain agent agent = create_react_agent(llm, tools)

Question 2 of 5

In the code review LangGraph workflow, what determines if suggestions are generated?

A The number of files changed in the diff
B The severity field — “request_changes” triggers the suggest_fixes node
C A random probability check for sampling
D The user must manually request suggestions
B — The severity field controls the conditional edge. The add_conditional_edges call checks state["severity"]. When it equals "request_changes", the graph routes to the suggest_fixes node. Otherwise it goes directly to END.
Question 3 of 5

Why is sandboxed execution important for AI dev tools?

A It makes code run faster by isolating processes
B AI-generated commands could damage files, exhaust resources, or access sensitive data
C It’s required by the MCP specification
D It prevents the LLM from hallucinating
B — AI-generated commands can be destructive. Sandboxing must prevent: file system damage (rm -rf), network abuse (curl/wget), infinite loops (timeout), and resource exhaustion (memory/output limits). The sandbox is a safety net between the AI and the system.
Question 4 of 5

A Store tool source code in a database
B Register, discover, and route tool calls dynamically
C Compile tools into faster machine code
D Authenticate users before tool access
B — Register, discover, and route dynamically. A tool registry enables dynamic discovery (search by tags or description) and routing (call by name) without hardcoding tool lists in every agent. New tools can be added at runtime.
